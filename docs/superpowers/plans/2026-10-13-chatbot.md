# Chatbot khách hàng (từ khoá + Gemini) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Khách đã đăng nhập chat với trợ lý ZomZop: khớp từ khoá admin soạn sẵn thì trả lời sẵn (kèm nút link), không khớp thì Gemini trả lời dựa trên danh sách món của quán.

**Architecture:** Bảng `chatbot_keywords` + model có hàm khớp từ khoá (bỏ dấu, theo ranh giới từ, ưu tiên từ khoá dài nhất). Service `App\Services\Chatbot` chọn từ khoá hoặc gọi Gemini REST qua `Http` facade (không thêm package), lỗi thì trả câu dự phòng. `ChatbotController` lưu lịch sử vào bảng có sẵn `ai_chat_sessions`. Giao diện là bóng chat trong `layouts/app.blade.php` + `resources/js/chatbot.js`. Admin quản lý từ khoá theo mẫu trang Banner.

**Tech Stack:** Laravel 12, PHP 8.3, Blade + Tailwind v4, Vite, JS thuần, PHPUnit (SQLite in-memory), Gemini API `v1beta/models/{model}:generateContent`.

**Spec:** `docs/superpowers/specs/2026-10-13-chatbot-design.md`

```php
<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Model;
use Illuminate\Support\Str;

/** Từ khoá → câu trả lời soạn sẵn của chatbot (admin quản lý) */
class ChatbotKeyword extends Model
{
    protected $fillable = ['keywords', 'reply', 'link_url', 'link_label', 'is_active'];

    protected $casts = ['is_active' => 'boolean'];

    /** "Cho XEM thực đơn?!" → "cho xem thuc don" */
    public static function normalize(string $text): string
    {
        return trim(preg_replace('/[^a-z0-9]+/', ' ', Str::lower(Str::ascii($text))));
    }

    /**
     * Dòng đang bật có từ khoá nằm trọn trong tin nhắn (theo ranh giới từ).
     * Nhiều dòng khớp → lấy dòng có từ khoá dài nhất (cụ thể nhất).
     * ponytail: quét cả bảng mỗi tin, ổn với vài chục từ khoá; nhiều hơn thì cache danh sách đã chuẩn hoá.
     */
    public static function findFor(string $message): ?self
    {
        $text = ' ' . static::normalize($message) . ' ';
        $best = null;
        $bestLength = 0;

        foreach (static::where('is_active', true)->get() as $row) {
            foreach (explode(',', $row->keywords) as $keyword) {
                $keyword = static::normalize($keyword);
                if ($keyword !== '' && strlen($keyword) > $bestLength && str_contains($text, " {$keyword} ")) {
                    $best = $row;
                    $bestLength = strlen($keyword);
                }
            }
        }

        return $best;
    }
}
```

```php
<?php

namespace App\Services;

use App\Models\ChatbotKeyword;
use App\Models\MenuItem;
use Illuminate\Support\Facades\Http;
use Illuminate\Support\Facades\Log;
use Illuminate\Support\Str;

/** Trả lời khách: từ khoá soạn sẵn trước, không khớp thì hỏi Gemini, lỗi thì câu dự phòng */
class Chatbot
{
    public const FALLBACK = 'Xin lỗi, mình đang bận một chút, bạn thử lại sau nhé.';

    public function __construct(private BranchMenu $menu)
    {
    }

    /**
     * @param  array<int, array{role: string, content: string}>  $history  tin cũ, chưa gồm $message
     * @return array{reply: string, link_url: ?string, link_label: ?string, source: string}
     */
    public function reply(string $message, array $history): array
    {
        if ($keyword = ChatbotKeyword::findFor($message)) {
            return [
                'reply'      => $keyword->reply,
                'link_url'   => $keyword->link_url,
                'link_label' => $keyword->link_label,
                'source'     => 'keyword',
            ];
        }

        $text = $this->askGemini($message, $history);

        return [
            'reply'      => $text ?? self::FALLBACK,
            'link_url'   => null,
            'link_label' => null,
            'source'     => $text === null ? 'fallback' : 'gemini',
        ];
    }

    private function askGemini(string $message, array $history): ?string
    {
        $key = config('services.gemini.key');
        if (!$key) {
            return null;
        }

        // Gemini cần lượt đầu là "user": bỏ các tin bot đứng đầu, bot → "model"
        $contents = collect($history)->takeLast(6)
            ->map(fn ($m) => ['role' => $m['role'] === 'user' ? 'user' : 'model', 'parts' => [['text' => $m['content']]]])
            ->skipUntil(fn ($c) => $c['role'] === 'user')
            ->push(['role' => 'user', 'parts' => [['text' => $message]]])
            ->values()->all();

        try {
            $response = Http::timeout(15)
                ->withHeaders(['x-goog-api-key' => $key])
                ->post('https://generativelanguage.googleapis.com/v1beta/models/' . config('services.gemini.model') . ':generateContent', [
                    'systemInstruction' => ['parts' => [['text' => $this->instruction()]]],
                    'contents'          => $contents,
                ]);
        } catch (\Throwable $e) {
            Log::warning('Chatbot: không gọi được Gemini', ['error' => $e->getMessage()]);

            return null;
        }

        $text = trim((string) $response->json('candidates.0.content.parts.0.text'));
        if ($response->failed() || $text === '') {
            Log::warning('Chatbot: Gemini trả lỗi hoặc không có chữ', ['status' => $response->status()]);

            return null;
        }

        return $text;
    }

    private function instruction(): string
    {
        $branchId = $this->menu->currentBranchId();
        $items = $this->menu->filterAvailable(MenuItem::query(), $branchId)->orderBy('menu_items.name')->get();
        $this->menu->applyPrices($items, $branchId);

        $menu = $items->map(fn (MenuItem $item) => "- {$item->name} — {$item->display_price} — "
            . Str::limit(strip_tags((string) $item->description), 120))->implode("\n");

        return <<<TEXT
        Bạn là trợ lý chăm sóc khách hàng của ZomZop, chuỗi cửa hàng đồ ăn nhanh tại Việt Nam.
        Trả lời bằng tiếng Việt, ngắn gọn, thân thiện, không dùng định dạng Markdown.
        Chỉ trả lời về món ăn, dinh dưỡng, cách đặt món và ZomZop. Câu hỏi ngoài phạm vi thì từ chối nhẹ nhàng và gợi ý hỏi về món ăn.
        Calo và dinh dưỡng là ước tính, hãy nói rõ là ước tính.
        Chỉ nhắc đến các món có trong danh sách dưới đây, không bịa món hay giá.

        Danh sách món đang bán (tên — giá — mô tả):
        {$menu}
        TEXT;
    }
}
```

```php
    {

            ['message' => ['required', 'string', 'max:500']],
            ['message.required' => 'Bạn chưa nhập tin nhắn.', 'message.max' => 'Tin nhắn tối đa 500 ký tự.']

    }
```

```js
// Bóng chat góc dưới phải: từ khoá soạn sẵn hoặc Gemini trả lời (POST /chatbot/message).
// Chèn chữ bằng textContent — không bao giờ innerHTML với tin của khách/Gemini.
const toggle = document.getElementById("chatbot-toggle");
const panel = document.getElementById("chatbot-panel");

if (toggle && !panel) {
    // Chưa đăng nhập: sang trang đăng nhập
    toggle.addEventListener("click", () => {
        window.location.href = document.querySelector('meta[name="login-url"]')?.content;
    });
}

if (toggle && panel) {
    const list = document.getElementById("chatbot-messages");
    const form = document.getElementById("chatbot-form");
    const input = form.querySelector('input[name="message"]');
    let loaded = false;

    const add = (role, text, linkUrl, linkLabel) => {
        const row = document.createElement("div");
        row.className = role === "user" ? "flex justify-end" : "flex justify-start";
        const bubble = document.createElement("div");
        bubble.className =
            "max-w-[80%] px-3 py-2 rounded-2xl whitespace-pre-line break-words " +
            (role === "user" ? "bg-red-500 text-white rounded-br-sm" : "bg-white text-slate-800 shadow-sm rounded-bl-sm");
        bubble.textContent = text;

        // Chỉ nhận đường dẫn trong web: "/..." nhưng không "//..."
        if (linkUrl && /^\/(?!\/)/.test(linkUrl)) {
            const link = document.createElement("a");
            link.href = linkUrl;
            link.textContent = linkLabel || "Xem";
            link.className = "mt-2 block w-fit rounded-full bg-red-50 px-3 py-1 text-xs font-semibold text-red-500 hover:bg-red-100";
            bubble.append(link);
        }

        row.append(bubble);
        list.append(row);
        list.scrollTop = list.scrollHeight;
        return bubble;
    };

    const open = () => {
        panel.classList.remove("hidden");
        input.focus();
        if (loaded) return;
        loaded = true;

        fetch("/chatbot/messages", { headers: { Accept: "application/json" } })
            .then((res) => res.json())
            .then((data) => {
                if (!data.messages.length) {
                    add("bot", "Xin chào! Mình là trợ lý ZomZop. Bạn muốn xem thực đơn, mã giảm giá hay hỏi gì về món ăn?");
                    return;
                }
                data.messages.forEach((m) => add(m.role, m.content, m.link_url, m.link_label));
            })
            .catch(() => {
                loaded = false; // mở lại sẽ thử tải lại
            });
    };
    const close = () => panel.classList.add("hidden");

    toggle.addEventListener("click", () => (panel.classList.contains("hidden") ? open() : close()));
    document.getElementById("chatbot-close").addEventListener("click", close);
    document.addEventListener("keydown", (e) => {
        if (e.key === "Escape") close();
    });

    form.addEventListener("submit", (e) => {
        e.preventDefault();
        const message = input.value.trim();
        if (!message) return;

        add("user", message);
        input.value = "";
        const pending = add("bot", "Đang trả lời…");

        fetch("/chatbot/message", {
            method: "POST",
            headers: {
                "Content-Type": "application/json",
                Accept: "application/json",
                "X-CSRF-TOKEN": document.querySelector('meta[name="csrf-token"]')?.content,
            },
            body: JSON.stringify({ message }),
        })
            .then((res) => {
                if (res.status === 429) throw new Error("Bạn gửi nhanh quá, đợi chút nhé.");
                if (!res.ok) throw new Error("Không gửi được tin nhắn, bạn thử lại nhé.");
                return res.json();
            })
            .then((data) => {
                pending.parentElement.remove();
                add("bot", data.reply, data.link_url, data.link_label);
            })
            .catch((err) => {
                pending.textContent = err instanceof TypeError ? "Mất kết nối, bạn thử lại nhé." : err.message;
            });
    });
```

```php
<?php

namespace App\Http\Controllers\Admin;

use App\Models\AdminLog;
use App\Models\ChatbotKeyword;
use Illuminate\Http\Request;

class ChatbotKeywordController extends AdminController
{
    private const MESSAGES = [
        'keywords.required'      => 'Vui lòng nhập ít nhất một từ khoá.',
        'reply.required'         => 'Vui lòng nhập câu trả lời.',
        'reply.max'              => 'Câu trả lời tối đa 1000 ký tự.',
        'link_url.regex'         => 'Link phải là trang trong web, bắt đầu bằng / (VD: /coupons).',
        'link_label.required_with' => 'Nhập chữ hiện trên nút khi có link.',
        'link_label.max'         => 'Chữ trên nút tối đa 50 ký tự.',
    ];

    private function payload(Request $request): array
    {
        $data = $request->validate([
            'keywords'   => ['required', 'string', 'max:255'],
            'reply'      => ['required', 'string', 'max:1000'],
            'link_url'   => ['nullable', 'string', 'max:255', 'regex:/^\/(?!\/)/'],
            'link_label' => ['nullable', 'required_with:link_url', 'string', 'max:50'],
        ], self::MESSAGES);
        $data['is_active'] = $request->boolean('is_active');
        if (empty($data['link_url'])) {
            $data['link_url'] = $data['link_label'] = null;
        }

        return $data;
    }

    public function index()
    {
        return view('admin.chatbot-keywords.index', ['keywords' => ChatbotKeyword::orderByDesc('id')->get()]);
    }

    public function create()
    {
        return view('admin.chatbot-keywords.form', ['keyword' => null]);
    }

    public function store(Request $request)
    {
        $data = $this->payload($request);
        $keyword = ChatbotKeyword::create($data);
        $this->log('chatbot_keyword.create', $keyword, ['after' => $data]);

        return redirect()->route('admin.chatbot-keywords.index')->with('success', 'Đã thêm từ khoá.');
    }

    public function edit(ChatbotKeyword $chatbotKeyword)
    {
        return view('admin.chatbot-keywords.form', ['keyword' => $chatbotKeyword]);
    }

    public function update(Request $request, ChatbotKeyword $chatbotKeyword)
    {
        $chatbotKeyword->fill($this->payload($request));
        $changes = AdminLog::diff($chatbotKeyword);
        $chatbotKeyword->save();
        $this->log('chatbot_keyword.update', $chatbotKeyword, $changes);

        return redirect()->route('admin.chatbot-keywords.index')->with('success', 'Đã cập nhật từ khoá.');
    }

    public function toggle(ChatbotKeyword $chatbotKeyword)
    {
        $chatbotKeyword->update(['is_active' => !$chatbotKeyword->is_active]);
        $this->log('chatbot_keyword.toggle', $chatbotKeyword, ['after' => ['is_active' => $chatbotKeyword->is_active]]);

        return back()->with('success', $chatbotKeyword->is_active ? 'Đã bật từ khoá.' : 'Đã tắt từ khoá.');
    }

    public function destroy(ChatbotKeyword $chatbotKeyword)
    {
        $chatbotKeyword->delete();
        $this->log('chatbot_keyword.delete', $chatbotKeyword, ['before' => $chatbotKeyword->only(['keywords', 'reply', 'link_url'])]);

        return back()->with('success', 'Đã xoá từ khoá.');
    }
}
```
